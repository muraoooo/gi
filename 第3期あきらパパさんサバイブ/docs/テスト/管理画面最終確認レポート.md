#### 1. API Route での権限チェック (`lib/admin.ts`)

```typescript
export async function checkAdminAuth(req, res) {
  const session = await getServerSession(req, res, authOptions);

  if (!session) {
    res.status(401).json({ error: '認証が必要です' }); // ← 未認証は401
    return null;
  }

  if (session.user?.role !== 'admin') {
    res.status(403).json({ error: '管理者権限が必要です' }); // ← 権限なしは403
    return null;
  }

  return { id, email, role };
}
```
